STEP 1 - DISCOVER ALL URL PATTERNS

Run this in the Django shell to get all URL patterns:

  python manage.py shell -c "
  from django.urls import get_resolver
  import json

  def collect_urls(resolver, prefix=''):
      urls = []
      for pattern in resolver.url_patterns:
          if hasattr(pattern, 'url_patterns'):
              urls.extend(collect_urls(pattern, prefix + str(pattern.pattern)))
          else:
              full = prefix + str(pattern.pattern)
              urls.append({'name': pattern.name, 'pattern': full})
      return urls

  print(json.dumps(collect_urls(get_resolver()), indent=2))
  "

From the output, build a concrete list of URLs to visit. Apply these filters:
- Skip patterns with required path parameters that cannot be filled with a sensible default (e.g. <pk>, <slug>) unless you can infer a real ID from the database.
- For parameterised URLs where a real object exists, query the database for the first available ID and include that URL once.
- Skip webhook, API-only, and export endpoints (patterns containing api/, webhook/, export/, .csv, .json).
- Include all auth pages (login, logout redirect, MFA flows, password reset).

Document the final URL list and the reason any URL was skipped.

STEP 3 - CREATE tests/bs3-bs5_migration/visual_baseline.spec.ts

Create this file, filling the PAGES array from the URLs discovered in Step 1:

  import { test, expect } from '@playwright/test';

  const VIEWPORTS = [
    { name: 'mobile',  width: 375,  height: 812  },
    { name: 'tablet',  width: 768,  height: 1024 },
    { name: 'desktop', width: 1280, height: 900  },
  ];

  const PAGES = [
    { name: 'home',  path: '/' },
    { name: 'login', path: '/accounts/login/' },
    // add every discovered URL here
  ];

  for (const viewport of VIEWPORTS) {
    test.describe(`${viewport.name} (${viewport.width}px)`, () => {
      test.use({ viewport: { width: viewport.width, height: viewport.height } });

      for (const page of PAGES) {
        test(`${page.name} - visual baseline`, async ({ page: pw }) => {
          await pw.goto(`http://localhost:8000${page.path}`);
          await pw.waitForLoadState('networkidle');
          const dismiss = pw.locator('[data-dismiss="modal"], .cookie-accept');
          if (await dismiss.count() > 0) await dismiss.first().click();
          await expect(pw).toHaveScreenshot(`${page.name}-${viewport.name}.png`, {
            maxDiffPixelRatio: 0.001,
            fullPage: true,
          });
        });
      }
    });
  }

STEP 4 - CREATE tests/bs3-bs5_migration/layout_assertions.spec.ts

  import { test, expect } from '@playwright/test';

  const PAGES = [ /* same list as Step 3 */ ];

  test.describe('Layout sanity - desktop', () => {
    test.use({ viewport: { width: 1280, height: 900 } });

    for (const page of PAGES) {
      test(`${page.name} - no horizontal overflow`, async ({ page: pw }) => {
        await pw.goto(`http://localhost:8000${page.path}`);
        await pw.waitForLoadState('networkidle');
        const scrollWidth = await pw.evaluate(() =>
          document.documentElement.scrollWidth > document.documentElement.clientWidth
        );
        expect(scrollWidth, 'Page has horizontal scroll').toBe(false);
      });
    }
  });

  test.describe('Layout sanity - mobile', () => {
    test.use({ viewport: { width: 375, height: 812 } });

    for (const page of PAGES) {
      test(`${page.name} - no horizontal overflow on mobile`, async ({ page: pw }) => {
        await pw.goto(`http://localhost:8000${page.path}`);
        await pw.waitForLoadState('networkidle');
        const scrollWidth = await pw.evaluate(() =>
          document.documentElement.scrollWidth > document.documentElement.clientWidth
        );
        expect(scrollWidth, 'Mobile layout has horizontal scroll').toBe(false);
      });
    }
  });

  test.describe('Navigation structure', () => {
    test.use({ viewport: { width: 1280, height: 900 } });

    test('navbar is above main content', async ({ page: pw }) => {
      await pw.goto('http://localhost:8000/');
      await pw.waitForLoadState('networkidle');
      const navBox  = await pw.locator('nav').first().boundingBox();
      const mainBox = await pw.locator('main, [role=main], .container').first().boundingBox();
      expect(navBox).toBeTruthy();
      expect(mainBox).toBeTruthy();
      expect(navBox!.y + navBox!.height).toBeLessThanOrEqual(mainBox!.y + 5);
    });

    test('navbar collapses on mobile', async ({ page: pw }) => {
      await pw.setViewportSize({ width: 375, height: 812 });
      await pw.goto('http://localhost:8000/');
      await pw.waitForLoadState('networkidle');
      const toggle = pw.locator('.navbar-toggle, .navbar-toggler');
      await expect(toggle).toBeVisible();
    });
  });
